## willingness

# FEATURE 1
```sql
SELECT CUST_INFO_ID, COUNT(*) AS TOTAL_PTP_COUNT, SUM(CASE WHEN PTP_STATUS = '2' THEN 1 ELSE 0 END) AS PTP_BROKEN_COUNT, SUM(CASE WHEN PTP_STATUS = '1' THEN 1 ELSE 0 END) AS PTP_KEPT_COUNT, ROUND(SUM(CASE WHEN PTP_STATUS = '2' THEN 1 ELSE 0 END) / NULLIF(COUNT(*), 0), 4) AS PTP_BROKEN_RATIO, ROUND(SYSDATE - MAX(CASE WHEN PTP_STATUS = '2' THEN PTP_DATE END), 0) AS DAYS_SINCE_LAST_BROKEN_PTP FROM PIO_COLL.PIO_PTP_COLLECTOR GROUP BY CUST_INFO_ID
```

# FEATURE 2 (Reachability (RPC Rate))
```sql
SELECT 
    CUST_INFO_ID,
    COUNT(*) AS TOTAL_ATTEMPTS_LIFETIME,

    SUM(CASE WHEN STATUS = 1 THEN 1 ELSE 0 END) AS SUCCESSFUL_CONTACTS,

    ROUND(
        CAST(SUM(CASE WHEN STATUS = 1 THEN 1 ELSE 0 END) AS FLOAT) 
        / NULLIF(COUNT(*), 0),
        4
    ) AS RPC_RATE
    FROM PIO_COLL.PIO_CONTACT_MANUAL_INFO
GROUP BY CUST_INFO_ID;
```

# FEATURE 3 (The "Ostrich" Signal (Recency))
```sql
select 
    CUST_INFO_ID,
ROUND(SYSDATE - MAX(CASE WHEN STATUS = 1 THEN UPDATED_DATE END), 0) AS DAYS_SINCE_LAST_RPC
from PIO_COLL.PIO_CONTACT_MANUAL_INFO
GROUP BY CUST_INFO_ID;
```

# FEATURE 4 (Intensity (Harassment Check))
```sql
select 
    CUST_INFO_ID,
SUM(CASE WHEN UPDATED_DATE > SYSDATE - 30 THEN 1 ELSE 0 END) AS ATTEMPTS_LAST_30D
```


# FEATURE 5 (Channel Preference)
```sql
SELECT 
    CUST_INFO_ID,
    SUM(CASE WHEN CONTACT_TYPE = 1 AND STATUS = 1 THEN 1 ELSE 0 END) AS SMS_DELIVERED_COUNT,
    SUM(CASE WHEN CONTACT_TYPE = 2 AND STATUS = 1 THEN 1 ELSE 0 END) AS CALL_ANSWERED_COUNT

FROM PIO_COLL.PIO_CONTACT_MANUAL_INFO
GROUP BY CUST_INFO_ID
```
